#####################################################################
# Eldritch Pacts - scripted localisation
#
#   [ROOT.GetEldritchGodName]    name of the country's patron god
#   [ROOT.GetEldritchPactStage]  current state of the pact
#   [ROOT.GetEldritchFavorLine]  favor read-out (empty before a pact)
#   [ROOT.GetEldritchProphetLine] the leader's standing with the patron
#                                 (empty before a pact)
#   [ROOT.GetEldritchMantleLine]  the Prophet's level and stats, for the
#                                 category header (empty before a pact)
#   [ROOT.GetEldritchMantleState] whether the mantle's gifts are in force
#####################################################################

defined_text = {
	name = GetEldritchPactStage

	text = {
		trigger = {
			has_country_flag = eldritch_pact_made
			check_variable = { eldritch_pact_tier > 2 }
		}
		localization_key = eldritch_stage_3
	}
	text = {
		trigger = {
			has_country_flag = eldritch_pact_made
			check_variable = { eldritch_pact_tier > 1 }
		}
		localization_key = eldritch_stage_2
	}
	text = {
		trigger = { has_country_flag = eldritch_pact_made }
		localization_key = eldritch_stage_1
	}
	text = {
		trigger = { has_country_flag = eldritch_ritual_underway }
		localization_key = eldritch_stage_ritual
	}
	text = {
		trigger = { has_country_flag = eldritch_lore_found }
		localization_key = eldritch_stage_lore
	}
	text = {
		localization_key = eldritch_stage_none
	}
}

defined_text = {
	name = GetEldritchFavorLine

	text = {
		trigger = { has_country_flag = eldritch_pact_made }
		localization_key = eldritch_favor_line
	}
	text = {
		localization_key = eldritch_favor_line_none
	}
}

defined_text = {
	name = GetEldritchProphetLine

	text = {
		trigger = {
			has_country_flag = eldritch_pact_made
			eldritch_leader_is_vessel = yes
		}
		localization_key = eldritch_prophet_line_vessel
	}
	text = {
		trigger = {
			has_country_flag = eldritch_pact_made
			eldritch_leader_is_chosen = yes
		}
		localization_key = eldritch_prophet_line_chosen
	}
	text = {
		trigger = {
			has_country_flag = eldritch_pact_made
			eldritch_leader_is_herald = yes
		}
		localization_key = eldritch_prophet_line_herald
	}
	text = {
		trigger = { has_country_flag = eldritch_pact_made }
		localization_key = eldritch_prophet_line_unmarked
	}
	text = {
		localization_key = eldritch_favor_line_none
	}
}

defined_text = {
	name = GetEldritchMantleLine

	text = {
		trigger = {
			has_country_flag = eldritch_pact_made
			has_variable = eldritch_prophet_level
			check_variable = { eldritch_prophet_insight > 0 }
		}
		localization_key = eldritch_mantle_line_insight
	}
	text = {
		trigger = {
			has_country_flag = eldritch_pact_made
			has_variable = eldritch_prophet_level
		}
		localization_key = eldritch_mantle_line
	}
	text = {
		localization_key = eldritch_favor_line_none
	}
}

defined_text = {
	name = GetEldritchMantleState

	text = {
		trigger = { eldritch_leader_is_herald = yes }
		localization_key = eldritch_mantle_state_worn
	}
	text = {
		localization_key = eldritch_mantle_state_unclaimed
	}
}
