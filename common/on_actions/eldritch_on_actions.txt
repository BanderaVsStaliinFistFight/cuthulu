#####################################################################
# Eldritch Pacts - on actions
#####################################################################

on_actions = {
	on_startup = {
		effect = {
			# Guarded by a global flag so the welcome is shown only once,
			# however many times on_startup is evaluated.
			if = {
				limit = { NOT = { has_global_flag = eldritch_welcome_shown } }
				set_global_flag = eldritch_welcome_shown
				every_country = {
					limit = { is_ai = no }
					country_event = { id = eldritch.2 days = 1 }
				}
			}
			# Followers of Hastur from saves made before the chapters had goals
			# get their chapter and buildings (also checked every month).
			every_country = {
				limit = { has_country_flag = eldritch_god_hastur }
				eldritch_carcosa_catch_up = yes
			}
			# Every nation's capital at the start of the game, for Chapter V of
			# the Road to Carcosa: a nation moves its capital when it falls, so
			# the goal asks for an old capital rather than the current one.
			if = {
				limit = { NOT = { has_global_flag = eldritch_capitals_recorded } }
				set_global_flag = eldritch_capitals_recorded
				every_country = {
					capital_scope = { set_state_flag = eldritch_original_capital }
				}
			}
		}
	}

	# The King's Play: ROOT is the state's new controller, FROM the old one.
	on_state_control_changed = {
		effect = {
			if = {
				limit = {
					has_country_flag = eldritch_carcosa_proclaimed
					FROM = { has_war_with = ROOT }
				}
				add_to_variable = { eldritch_dread = 3 }
				eldritch_crisis_check_acts = yes
			}
			if = {
				limit = { FROM = { has_country_flag = eldritch_carcosa_proclaimed } }
				FROM = {
					subtract_from_variable = { eldritch_dread = 3 }
					eldritch_crisis_clamp_dread = yes
				}
			}
		}
	}

	# The King's Play: ROOT has capitulated to FROM.
	on_capitulation = {
		effect = {
			if = {
				limit = { FROM = { has_country_flag = eldritch_carcosa_proclaimed } }
				FROM = {
					add_to_variable = { eldritch_dread = 25 }
					eldritch_crisis_check_acts = yes
				}
			}
			if = {
				limit = { has_country_flag = eldritch_carcosa_proclaimed }
				divide_variable = { eldritch_dread = 2 }
			}
		}
	}
}
