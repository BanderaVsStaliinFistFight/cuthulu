# Runs every 30 days for pact holders (hidden event eldritch.100).
eldritch_monthly_pulse = {
	# pacts sealed before the Prophet's mantle existed get one now, and
	# older mantles get their skill check odds
	eldritch_prophet_init = yes
	eldritch_prophet_update_mantle = yes

	# Apostolic Succession: the god marks each new leader itself
	if = {
		limit = {
			has_country_flag = eldritch_perk_will_3
			NOT = { eldritch_leader_is_herald = yes }
		}
		eldritch_mark_herald_effect = yes
	}

	if = {
		limit = { check_variable = { eldritch_pact_tier > 2 } }
		add_to_variable = { eldritch_favor = 12 }
	}
	else_if = {
		limit = { check_variable = { eldritch_pact_tier > 1 } }
		add_to_variable = { eldritch_favor = 8 }
	}
	else = {
		add_to_variable = { eldritch_favor = 5 }
	}
	if = {
		limit = { has_country_flag = eldritch_great_temple }
		add_to_variable = { eldritch_favor = 3 }
	}
	eldritch_clamp_favor = yes

	# While the god walks the earth our armies take heart and our enemies
	# despair; once its division is destroyed the spirit goes.
	if = {
		limit = { eldritch_god_incarnate_deployed = yes }
		eldritch_god_walks_effect = yes
	}
	else_if = {
		limit = { has_idea = eldritch_god_walks }
		remove_ideas = eldritch_god_walks
	}

	# The gods are not silent partners.
	random_list = {
		88 = { }
		3 = { country_event = { id = eldritch.40 days = 2 } }
		3 = { country_event = { id = eldritch.41 days = 2 } }
		3 = { country_event = { id = eldritch.42 days = 2 } }
		3 = { country_event = { id = eldritch.43 days = 2 } }
	}
}

# Our armies fight beside a god; every nation at war with us is struck with
# dread for the next 35 days. Renewed by the monthly pulse while the god's
# division lives.
eldritch_god_walks_effect = {
	if = {
		limit = { NOT = { has_idea = eldritch_god_walks } }
		add_ideas = eldritch_god_walks
	}
	custom_effect_tooltip = eldritch_dread_tt
	hidden_effect = {
		every_country = {
			limit = { has_war_with = ROOT }
			if = {
				limit = { has_idea = eldritch_dread_of_the_god }
				remove_ideas = eldritch_dread_of_the_god
			}
			add_timed_idea = { idea = eldritch_dread_of_the_god days = 35 }
		}
	}
}

# Field marshals are chosen before ordinary generals.
eldritch_anoint_general_effect = {
	if = {
		limit = {
			any_army_leader = {
				is_field_marshal = yes
				NOT = { has_unit_leader_flag = eldritch_anointed }
				NOT = { has_trait = eldritch_avatar_land }
			}
		}
		random_army_leader = {
			limit = {
				is_field_marshal = yes
				NOT = { has_unit_leader_flag = eldritch_anointed }
				NOT = { has_trait = eldritch_avatar_land }
			}
			eldritch_bless_army_leader = yes
		}
	}
	else = {
		random_army_leader = {
			limit = {
				NOT = { has_unit_leader_flag = eldritch_anointed }
				NOT = { has_trait = eldritch_avatar_land }
			}
			eldritch_bless_army_leader = yes
		}
	}
}

#####################################################################
# THE PROPHET'S MANTLE - the character sheet
#
# The mantle belongs to the cult, not to one leader: whoever bears the
# god's mark (the Herald trait) wears it, and its bonuses lapse while the
# leader is unmarked. Country variables:
#   eldritch_prophet_level     level, from 1
#   eldritch_prophet_xp        experience towards the next level
#   eldritch_prophet_xp_next   experience the next level needs (100, +50
#                              per level)
#   eldritch_prophet_insight   unspent points
#   eldritch_prophet_lore / _zeal / _will
#                              stats, 1 to 10
#   eldritch_prophet_<stat>_bonus / _pct
#                              the stat as a modifier (x 0.02) and as a
#                              percentage for the sheet (x 2)
#####################################################################

eldritch_prophet_init = {
	if = {
		limit = { NOT = { has_variable = eldritch_prophet_level } }
		set_variable = { eldritch_prophet_level = 1 }
		set_variable = { eldritch_prophet_xp = 0 }
		set_variable = { eldritch_prophet_xp_next = 100 }
		set_variable = { eldritch_prophet_insight = 0 }
		set_variable = { eldritch_prophet_lore = 1 }
		set_variable = { eldritch_prophet_zeal = 1 }
		set_variable = { eldritch_prophet_will = 1 }
		eldritch_prophet_update_mantle = yes
		add_dynamic_modifier = { modifier = eldritch_prophet_mantle }
	}
}

eldritch_prophet_update_mantle = {
	set_variable = { eldritch_prophet_lore_bonus = eldritch_prophet_lore }
	multiply_variable = { eldritch_prophet_lore_bonus = 0.02 }
	set_variable = { eldritch_prophet_zeal_bonus = eldritch_prophet_zeal }
	multiply_variable = { eldritch_prophet_zeal_bonus = 0.02 }
	set_variable = { eldritch_prophet_will_bonus = eldritch_prophet_will }
	multiply_variable = { eldritch_prophet_will_bonus = 0.02 }
	set_variable = { eldritch_prophet_lore_pct = eldritch_prophet_lore }
	multiply_variable = { eldritch_prophet_lore_pct = 2 }
	set_variable = { eldritch_prophet_zeal_pct = eldritch_prophet_zeal }
	multiply_variable = { eldritch_prophet_zeal_pct = 2 }
	set_variable = { eldritch_prophet_will_pct = eldritch_prophet_will }
	multiply_variable = { eldritch_prophet_will_pct = 2 }

	# Skill check odds, in percent (5 to 95). See SKILL CHECKS below.
	set_variable = { eldritch_odds_commune = eldritch_prophet_will }
	multiply_variable = { eldritch_odds_commune = 4 }
	add_to_variable = { eldritch_odds_commune = 60 }
	set_variable = { eldritch_odds_curse = eldritch_prophet_lore }
	multiply_variable = { eldritch_odds_curse = 6 }
	add_to_variable = { eldritch_odds_curse = 45 }
	set_variable = { eldritch_odds_preach = eldritch_prophet_zeal }
	multiply_variable = { eldritch_odds_preach = 6 }
	add_to_variable = { eldritch_odds_preach = 40 }
	set_variable = { eldritch_odds_incarnate = eldritch_prophet_zeal }
	multiply_variable = { eldritch_odds_incarnate = 6 }
	add_to_variable = { eldritch_odds_incarnate = 30 }
	set_variable = { eldritch_odds_vessel = eldritch_prophet_will }
	multiply_variable = { eldritch_odds_vessel = 6 }
	add_to_variable = { eldritch_odds_vessel = 30 }
	# perks of the Prophet's Path
	if = {
		limit = { has_country_flag = eldritch_perk_lore_1 }
		add_to_variable = { eldritch_odds_curse = 10 }
	}
	if = {
		limit = { has_country_flag = eldritch_perk_zeal_2 }
		add_to_variable = { eldritch_odds_preach = 10 }
		add_to_variable = { eldritch_odds_incarnate = 10 }
	}
	if = {
		limit = { has_country_flag = eldritch_perk_will_1 }
		add_to_variable = { eldritch_odds_commune = 10 }
		add_to_variable = { eldritch_odds_vessel = 10 }
		add_to_variable = { eldritch_prophet_will_bonus = 0.03 }
		add_to_variable = { eldritch_prophet_will_pct = 3 }
	}
	clamp_variable = { var = eldritch_odds_commune min = 5 max = 95 }
	clamp_variable = { var = eldritch_odds_curse min = 5 max = 95 }
	clamp_variable = { var = eldritch_odds_preach min = 5 max = 95 }
	clamp_variable = { var = eldritch_odds_incarnate min = 5 max = 95 }
	clamp_variable = { var = eldritch_odds_vessel min = 5 max = 95 }
}

# Spends enough experience for as many levels as it covers; each level
# brings a point of Insight. Announces the new level (eldritch.81).
eldritch_prophet_check_level = {
	set_temp_variable = { eldritch_levels_gained = 0 }
	while_loop_effect = {
		limit = {
			check_variable = {
				var = eldritch_prophet_xp
				value = eldritch_prophet_xp_next
				compare = greater_than_or_equals
			}
		}
		subtract_from_variable = { eldritch_prophet_xp = eldritch_prophet_xp_next }
		add_to_variable = { eldritch_prophet_level = 1 }
		add_to_variable = { eldritch_prophet_insight = 1 }
		add_to_variable = { eldritch_prophet_xp_next = 50 }
		add_to_temp_variable = { eldritch_levels_gained = 1 }
	}
	if = {
		limit = { check_variable = { eldritch_levels_gained > 0 } }
		country_event = { id = eldritch.81 }
	}
}

#####################################################################
# SKILL CHECKS
#
# A check rolls d100 against odds that grow with one of the Prophet's
# stats (eldritch_odds_<check>, kept by eldritch_prophet_update_mantle):
#
#   check       stat   odds                  where
#   commune     Will   60% + 4% per point    Commune: keep the Prophet's mind
#   curse       Lore   45% + 6% per point    the god's curses land in full
#   preach      Zeal   40% + 6% per point    Preach stirs the crowd further
#   incarnate   Zeal   30% + 6% per point    the god is pleased to be called
#   vessel      Will   30% + 6% per point    the Vessel keeps more of its mind
#
# Each check sets these country variables, which the result text shows:
#   eldritch_check_stat    1 Lore, 2 Zeal, 3 Will
#   eldritch_check_kind    1 commune, 2 curse, 3 preach, 4 incarnate, 5 vessel
#   eldritch_check_chance  the odds rolled against
#   eldritch_check_roll    the roll, 0 to 99
#   eldritch_check_passed  1 on a success, 0 on a failure
#
# Stats count in checks even while the mantle is unclaimed: the cult
# keeps what it has learned, even when the god's gifts lapse.
#####################################################################

# Rolls against eldritch_check_chance.
eldritch_roll_check = {
	set_variable_to_random = {
		var = eldritch_check_roll
		min = 0
		max = 100
		integer = yes
	}
	if = {
		limit = {
			check_variable = {
				var = eldritch_check_roll
				value = eldritch_check_chance
				compare = less_than
			}
		}
		set_variable = { eldritch_check_passed = 1 }
	}
	else = {
		set_variable = { eldritch_check_passed = 0 }
	}
}

#####################################################################
# THE ROAD TO CARCOSA - opening a chapter
#
# Shows its decision, records the conquest target where it has one
# (states controlled now + N), unlocks the chapter's building, and
# for the AI, which would not build it, puts the building down.
#####################################################################

eldritch_carcosa_open_1 = {
	set_country_flag = eldritch_carcosa_1_open
	set_technology = { eldritch_carcosa_theatre = 1 popup = no }
	if = {
		limit = { is_ai = yes }
		capital_scope = { add_building_construction = { type = eldritch_yellow_theatre level = 1 instant_build = yes } }
	}
}

eldritch_carcosa_open_2 = {
	set_country_flag = eldritch_carcosa_2_open
	if = {
		limit = { is_ai = yes }
		random_owned_controlled_state = {
			limit = { NOT = { eldritch_yellow_theatre > 0 } }
			add_building_construction = { type = eldritch_yellow_theatre level = 1 instant_build = yes }
		}
		random_owned_controlled_state = {
			limit = { NOT = { eldritch_yellow_theatre > 0 } }
			add_building_construction = { type = eldritch_yellow_theatre level = 1 instant_build = yes }
		}
	}
}
