#####################################################################
# Eldritch Pacts - decisions
#
# 1. Seek the Forbidden Lore         unlocks the summoning rituals
# 2. Summon <god>                    choose ONE patron, ever
# 3. Common rites                    earn Favor, deepen the pact,
#                                    anoint leaders, call servitors
# 4. Patron-specific blessings       timed blessing, avatar, unique rite
# 5. The Prophet                     rites the country leader performs in
#                                    person; most need the leader to bear
#                                    the patron's Herald trait
#
# Favor costs use custom_cost_trigger + custom_cost_text and are paid
# in complete_effect.
#####################################################################

			check_variable = { eldritch_favor > 99 }

		custom_cost_text = eldritch_cost_favor_100
		fire_only_once = yes

		visible = {
			has_country_flag = eldritch_carcosa_4_open
			NOT = { has_country_flag = eldritch_carcosa_4_done }
		}

		available = {
			eldritch_pact_tier_3 = yes
			custom_trigger_tooltip = {
				tooltip = eldritch_carcosa_goal_observatories_tt

		complete_effect = {
			hidden_effect = {

			eldritch_prophet_gain_xp_100 = yes
			eldritch_prophet_gain_insight = yes
			custom_effect_tooltip = eldritch_carcosa_opens_5_tt
			hidden_effect = { eldritch_carcosa_open_5 = yes }
		}

			factor = 10
		}
	}

			check_variable = { eldritch_favor > 149 }
		}
		custom_cost_text = eldritch_cost_favor_150
		fire_only_once = yes

		visible = {

			NOT = { has_country_flag = eldritch_carcosa_5_done }

			custom_trigger_tooltip = {
				tooltip = eldritch_carcosa_goal_states_5_tt

			custom_trigger_tooltip = {
				tooltip = eldritch_carcosa_goal_capital_tt

				set_variable = { eldritch_carcosa_chapter = 5 }

	# Chapter 6: opens when chapter 5 is complete; the goals are
	# the available block.
	eldritch_carcosa_6 = {
		icon = eldritch_hastur
		custom_cost_trigger = {
			check_variable = { eldritch_favor > 199 }
		}
		custom_cost_text = eldritch_cost_favor_200
		fire_only_once = yes

		visible = {
			has_country_flag = eldritch_carcosa_6_open
			NOT = { has_country_flag = eldritch_carcosa_6_done }
		}

		available = {
			custom_trigger_tooltip = {
				tooltip = eldritch_carcosa_goal_palace_tt
				capital_scope = { eldritch_yellow_palace > 0 }
			}
			custom_trigger_tooltip = {
				tooltip = eldritch_carcosa_goal_capitulated_tt
				any_country = {
					has_war_with = ROOT
					has_capitulated = yes
				}
			}
			custom_trigger_tooltip = {
				tooltip = eldritch_carcosa_goal_prophet_tt
				OR = {
					has_country_leader_with_trait = eldritch_hastur_vessel
					eldritch_leader_is_chosen = yes
				}
			}
		}

		complete_effect = {
			hidden_effect = {
				subtract_from_variable = { eldritch_favor = 200 }
				set_country_flag = eldritch_carcosa_6_done
				set_variable = { eldritch_carcosa_chapter = 6 }
				country_event = { id = eldritch.362 }

			custom_effect_tooltip = eldritch_carcosa_ready_tt
			hidden_effect = { set_country_flag = eldritch_carcosa_ready }

			eldritch_prophet_gain_xp_100 = yes
			eldritch_prophet_gain_insight = yes

			factor = 10
		}

	# The end of the road: the nation becomes Carcosa.
	eldritch_proclaim_carcosa = {
		icon = eldritch_carcosa
		cost = 100
		fire_only_once = yes

		visible = {
			has_country_flag = eldritch_carcosa_ready
		}

		available = {
			has_stability > 0.25
		}

		complete_effect = {
			set_cosmetic_tag = ELDRITCH_CARCOSA
			custom_effect_tooltip = eldritch_proclaim_carcosa_tt
			if = {
				limit = { has_idea = eldritch_carcosa_road_5 }
				swap_ideas = { remove_idea = eldritch_carcosa_road_5 add_idea = eldritch_yellow_throne }
			}
			else = {
				add_ideas = eldritch_yellow_throne
			}
			eldritch_prophet_gain_xp_100 = yes
			hidden_effect = {
				set_country_flag = eldritch_carcosa_proclaimed
				country_event = { id = eldritch.370 }
				every_other_country = {
					news_event = { id = eldritch_news.5 hours = 6 }
				}
			}
		}

		ai_will_do = {
			factor = 10
		}
	}
