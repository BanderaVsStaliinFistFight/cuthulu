#   eldritch.40-43    the patron's whims (random, monthly)
#   eldritch.60-62    received by the target of a curse

#   eldritch_news.1/2 world news when a god is summoned / ascends

#####################################################################
# THE WHIMS OF THE PATRON (fired at random by eldritch_monthly_pulse)
#####################################################################

# Tribute demanded
country_event = {
	id = eldritch.40
	title = eldritch.40.t
	desc = eldritch.40.d
	picture = GFX_report_event_eldritch_omen

	is_triggered_only = yes

	option = {
		name = eldritch.40.a
		trigger = { has_manpower > 9999 }
		add_manpower = -10000
		eldritch_gain_favor_25 = yes
		ai_chance = { factor = 50 }
	}

	option = {
		name = eldritch.40.b
		add_political_power = -50
		eldritch_gain_favor_15 = yes
		ai_chance = { factor = 40 }
	}

	option = {
		name = eldritch.40.c
		eldritch_lose_favor_40 = yes
		add_stability = -0.05
		ai_chance = { factor = 10 }
	}
}

#####################################################################
# MONTHLY PULSE - hidden, re-schedules itself every 30 days.
# Started when a pact is sealed (eldritch_seal_pact_common).
#####################################################################

country_event = {
	id = eldritch.100
	hidden = yes

	is_triggered_only = yes

	immediate = {
		if = {
			limit = { has_country_flag = eldritch_pact_made }
			eldritch_monthly_pulse = yes
			country_event = { id = eldritch.100 days = 30 }
		}
	}
}
